cmake_minimum_required(VERSION 3.20)
project(TkaInstallerLauncher LANGUAGES CXX)
add_executable(TkaInstaller WIN32 main.cpp setup.rc)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../../VERSION" TKA_VERSION LIMIT_COUNT 1)
if(NOT TKA_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "Invalid VERSION file")
endif()
string(REPLACE "." ";" TKA_VERSION_PARTS "${TKA_VERSION}")
list(GET TKA_VERSION_PARTS 0 TKA_VERSION_MAJOR)
list(GET TKA_VERSION_PARTS 1 TKA_VERSION_MINOR)
list(GET TKA_VERSION_PARTS 2 TKA_VERSION_PATCH)
configure_file(setup-version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/setup-version.rc" @ONLY)
target_sources(TkaInstaller PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/setup-version.rc")
target_include_directories(TkaInstaller PRIVATE ../Tka.Installer/Assets)
target_compile_features(TkaInstaller PRIVATE cxx_std_17)
target_compile_definitions(TkaInstaller PRIVATE UNICODE _UNICODE)
set_property(TARGET TkaInstaller PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_target_properties(TkaInstaller PROPERTIES OUTPUT_NAME "Setup Techno Kitten Adventure")
add_executable(TkaGameLauncher WIN32 game.cpp game.rc)
target_include_directories(TkaGameLauncher PRIVATE ../Tka.Host/Assets)
target_compile_features(TkaGameLauncher PRIVATE cxx_std_17)
target_compile_definitions(TkaGameLauncher PRIVATE UNICODE _UNICODE)
set_property(TARGET TkaGameLauncher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_target_properties(TkaGameLauncher PROPERTIES OUTPUT_NAME "Techno Kitten Adventure")
